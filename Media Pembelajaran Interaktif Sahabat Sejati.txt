// --- Web Audio Helper for Sound Effects ---
const playSound = (type) => {
  try {
    const ctx = new (window.AudioContext || window.webkitAudioContext)();
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.connect(gain);
    gain.connect(ctx.destination);

    if (type === 'correct') {
      osc.type = 'triangle';
      osc.frequency.setValueAtTime(523.25, ctx.currentTime); // C5
      osc.frequency.exponentialRampToValueAtTime(659.25, ctx.currentTime + 0.1); // E5
      osc.frequency.exponentialRampToValueAtTime(783.99, ctx.currentTime + 0.2); // G5
      gain.gain.setValueAtTime(0.3, ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.35);
      osc.start();
      osc.stop(ctx.currentTime + 0.35);
    } else if (type === 'wrong') {
      osc.type = 'sawtooth';
      osc.frequency.setValueAtTime(250, ctx.currentTime);
      osc.frequency.exponentialRampToValueAtTime(150, ctx.currentTime + 0.2);
      gain.gain.setValueAtTime(0.3, ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.25);
      osc.start();
      osc.stop(ctx.currentTime + 0.25);
    } else if (type === 'win') {
      const notes = [523.25, 659.25, 783.99, 1046.50];
      notes.forEach((freq, idx) => {
        const o = ctx.createOscillator();
        const g = ctx.createGain();
        o.connect(g);
        g.connect(ctx.destination);
        o.frequency.value = freq;
        g.gain.setValueAtTime(0.25, ctx.currentTime + idx * 0.1);
        g.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + idx * 0.1 + 0.2);
        o.start(ctx.currentTime + idx * 0.1);
        o.stop(ctx.currentTime + idx * 0.1 + 0.25);
      });
    } else if (type === 'click') {
      osc.type = 'sine';
      osc.frequency.setValueAtTime(400, ctx.currentTime);
      gain.gain.setValueAtTime(0.15, ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.08);
      osc.start();
      osc.stop(ctx.currentTime + 0.08);
    }
  } catch (e) {
    // Audio context not allowed or supported without user gesture
  }
};

/* =========================================================================
   TAB 3: MINI GAME 2 - PASANGKAN PERBUATAN SAHABAT
========================================================================= */
function SectionGameMatching({ setTotalStars, triggerSound }) {
  const [score, setScore] = useState(0);
  const [completed, setCompleted] = useState(false);

  const items = [
    {
      id: 1,
      situation: "Teman sedang sakit dirawat di rumah",
      goodAction: "Menjenguk dan mendoakan kesembuhannya",
      badAction: "Membiarkan dan meninggalkannya sendirian",
    },
    {
      id: 2,
      situation: "Teman sedang bersedih atau menangis",
      goodAction: "Menghibur dan mendengarkan ceritanya",
      badAction: "Ejek dan tertawa terbahak-bahak",
    },
    {
      id: 3,
      situation: "Teman kesulitan belajar materi kelas",
      goodAction: "Belajar bersama dan saling membantu",
      badAction: "Pelit ilmu dan tidak mau bantu",
    },
    {
      id: 4,
      situation: "Teman berbeda agama atau suku",
      goodAction: "Tetap berteman baik dan saling menghormati",
      badAction: "Jauhi dan hanya berteman yang sama saja",
    }
  ];

  const [answers, setAnswers] = useState({});

  const handleSelect = (itemId, choice) => {
    if (answers[itemId]) return; // process only once per item

    const isCorrect = choice === 'good';
    setAnswers(prev => ({ ...prev, [itemId]: choice }));

    if (isCorrect) {
      triggerSound('correct');
      const newScore = score + 1;
      setScore(newScore);

      if (newScore === items.length) {
        triggerSound('win');
        setCompleted(true);
        setTotalStars(prev => prev + 3);
      }
    } else {
      triggerSound('wrong');
    }
  };

  const resetGame = () => {
    setAnswers({});
    setScore(0);
    setCompleted(false);
  };

  return (
    <div className="bg-white rounded-3xl p-6 shadow-xl border-4 border-emerald-300">
      <div className="text-center mb-6">
        <h2 className="text-2xl font-black text-emerald-900 flex items-center justify-center gap-2">
          🧩 Pilih Perbuatan Sahabat Sejati
        </h2>
        <p className="text-xs text-emerald-700 font-bold mt-1">
          Pilih sikap anak Tuhan yang benar untuk setiap situasi di bawah ini!
        </p>
      </div>

      <div className="space-y-4">
        {items.map((item) => {
          const isAnswered = answers[item.id];
          return (
            <div key={item.id} className="bg-emerald-50/60 p-4 rounded-2xl border-2 border-emerald-200 space-y-3">
              <div className="flex items-center space-x-2">
                <span className="bg-emerald-500 text-white w-6 h-6 rounded-full flex items-center justify-center font-extrabold text-xs">
                  {item.id}
                </span>
                <h4 className="font-extrabold text-slate-800 text-sm">{item.situation}</h4>
              </div>

              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                {/* Option Good */}
                <button
                  onClick={() => handleSelect(item.id, 'good')}
                  disabled={!!isAnswered}
                  className={`p-3 rounded-xl border-2 text-left text-xs font-bold transition-all flex items-center justify-between ${
                    isAnswered === 'good'
                      ? 'bg-emerald-500 text-white border-emerald-600 shadow'
                      : isAnswered
                      ? 'bg-slate-100 text-slate-400 border-slate-200'
                      : 'bg-white hover:bg-emerald-100 text-emerald-900 border-emerald-300'
                  }`}
                >
                  <span>{item.goodAction}</span>
                  {isAnswered === 'good' && <CheckCircle2 className="w-5 h-5 text-white flex-shrink-0" />}
                </button>

                {/* Option Bad */}
                <button
                  onClick={() => handleSelect(item.id, 'bad')}
                  disabled={!!isAnswered}
                  className={`p-3 rounded-xl border-2 text-left text-xs font-bold transition-all flex items-center justify-between ${
                    isAnswered === 'bad'
                      ? 'bg-rose-500 text-white border-rose-600 shadow'
                      : isAnswered
                      ? 'bg-slate-100 text-slate-400 border-slate-200'
                      : 'bg-white hover:bg-rose-100 text-rose-900 border-rose-200'
                  }`}
                >
                  <span>{item.badAction}</span>
                  {isAnswered === 'bad' && <XCircle className="w-5 h-5 text-white flex-shrink-0" />}
                </button>
              </div>
            </div>
          );
        })}
      </div>

      {completed && (
        <div className="mt-6 p-5 bg-gradient-to-r from-emerald-400 to-teal-500 text-white rounded-2xl text-center shadow-lg animate-bounce">
          <h3 className="text-xl font-black">Luar Biasa! Kamu Adalah Sahabat yang Baik! 🌟</h3>
          <p className="text-xs font-bold mt-1">Kamu berhasil memilih semua tindakan penuh kasih!</p>
          <div className="mt-3 inline-flex items-center space-x-2 bg-white text-emerald-800 px-4 py-1.5 rounded-full font-black text-xs">
            <Star className="w-4 h-4 fill-yellow-400 text-yellow-500" />
            <span>+3 Bintang Diterima</span>
          </div>
          <button
            onClick={resetGame}
            className="block mx-auto mt-4 px-5 py-2 bg-white text-emerald-700 hover:bg-emerald-50 font-black rounded-full text-xs shadow"
          >
            Coba Lagi
          </button>
        </div>
      )}
    </div>
  );
}

/* =========================================================================
   TAB 4: KUIS INTERAKTIF (5 Questions + Scoring)
========================================================================= */
function SectionKuis({ setTotalStars, triggerSound }) {
  const [currentQ, setCurrentQ] = useState(0);
  const [selectedOpt, setSelectedOpt] = useState(null);
  const [userAnswers, setUserAnswers] = useState([]);
  const [isFinished, setIsFinished] = useState(false);

  const questions = [
    {
      q: "Menurut Amsal 17:17, seorang sahabat menaruh kasih pada saat...",
      options: ["Saat senang saja", "Setiap waktu", "Saat diberi hadiah saja", "Saat dipuji"],
      answer: 1
    },
    {
      q: "Manusia disebut makhluk sosial artinya...",
      options: [
        "Manusia bisa hidup sendiri tanpa bantuan",
        "Manusia suka tinggal di hutan",
        "Manusia selalu membutuhkan orang lain",
        "Manusia tidak butuh teman"
      ],
      answer: 2
    },
    {
      q: "Di kota manakah Yesus mengajar saat 4 sahabat membongkar atap?",
      options: ["Yerusalem", "Kapernaum", "Nazaret", "Betlehem"],
      answer: 1
    },
    {
      q: "Mengapa kerabat dan sahabat menurunkan si lumpuh dari atap rumah?",
      options: [
        "Karena pintu rumah sudah sangat penuh orang",
        "Karena mereka ingin bermain tali",
        "Karena rumahnya tidak memiliki pintu",
        "Karena disuruh oleh pemilik rumah"
      ],
      answer: 0
    },
    {
      q: "Bagaimana cara kita mengasihi teman di sekolah?",
      options: [
        "Hanya memilih teman yang kaya saja",
        "Saling menolong, mendoakan, dan tidak membeda-bedakan",
        "Memusuhi teman yang berbeda agama",
        "Mengejek teman yang sedang sedih"
      ],
      answer: 1
    }
  ];

  const handleSelectOption = (index) => {
    setSelectedOpt(index);
    triggerSound('click');
  };

  const handleNext = () => {
    if (selectedOpt === null) return;

    const isCorrect = selectedOpt === questions[currentQ].answer;
    if (isCorrect) triggerSound('correct');
    else triggerSound('wrong');

    const updatedAnswers = [...userAnswers, { question: currentQ, selected: selectedOpt, isCorrect }];
    setUserAnswers(updatedAnswers);

    if (currentQ + 1 < questions.length) {
      setCurrentQ(currentQ + 1);
      setSelectedOpt(null);
    } else {
      triggerSound('win');
      setIsFinished(true);
      const correctCount = updatedAnswers.filter(a => a.isCorrect).length;
      setTotalStars(prev => prev + correctCount * 2);
    }
  };

  const resetKuis = () => {
    setCurrentQ(0);
    setSelectedOpt(null);
    setUserAnswers([]);
    setIsFinished(false);
  };

  const calculateScore = () => {
    const correctCount = userAnswers.filter(a => a.isCorrect).length;
    return Math.round((correctCount / questions.length) * 100);
  };

  return (
    <div className="bg-white rounded-3xl p-6 shadow-xl border-4 border-purple-300">
      {!isFinished ? (
        <div>
          {/* Header quiz info */}
          <div className="flex items-center justify-between mb-4 pb-2 border-b-2 border-slate-100">
            <div>
              <h2 className="text-xl font-black text-purple-900">📝 Kuis Sahabat Sejati</h2>
              <p className="text-xs font-bold text-purple-600">Jawab pertanyaan berikut dengan teliti!</p>
            </div>
            <span className="text-xs font-black bg-purple-100 text-purple-700 px-3 py-1.5 rounded-full">
              Soal {currentQ + 1} / {questions.length}
            </span>
          </div>

          {/* Progress bar */}
          <div className="w-full bg-slate-100 h-3 rounded-full overflow-hidden mb-6">
            <div
              className="bg-purple-600 h-full transition-all duration-300"
              style={{ width: `${((currentQ + 1) / questions.length) * 100}%` }}
            />
          </div>

          {/* Question content */}
          <div className="bg-purple-50/60 p-5 rounded-2xl border-2 border-purple-200 mb-6">
            <h3 className="text-base sm:text-lg font-black text-purple-950">
              {questions[currentQ].q}
            </h3>
          </div>

          {/* Options */}
          <div className="space-y-3">
            {questions[currentQ].options.map((opt, idx) => (
              <button
                key={idx}
                onClick={() => handleSelectOption(idx)}
                className={`w-full text-left p-4 rounded-2xl border-2 font-bold text-xs sm:text-sm transition-all flex items-center justify-between ${
                  selectedOpt === idx
                    ? 'bg-purple-600 text-white border-purple-700 shadow-md transform scale-[1.01]'
                    : 'bg-white hover:bg-purple-50 text-slate-800 border-purple-200'
                }`}
              >
                <span>{opt}</span>
                {selectedOpt === idx && <CheckCircle2 className="w-5 h-5 text-white" />}
              </button>
            ))}
          </div>

          {/* Submit/Next button */}
          <div className="mt-6 flex justify-end">
            <button
              disabled={selectedOpt === null}
              onClick={handleNext}
              className={`px-6 py-3 rounded-full font-black text-sm flex items-center gap-2 transition-all ${
                selectedOpt === null
                  ? 'bg-slate-200 text-slate-400 cursor-not-allowed'
                  : 'bg-purple-600 hover:bg-purple-700 text-white shadow-lg'
              }`}
            >
              {currentQ + 1 === questions.length ? 'Selesai & Lihat Hasil 🏆' : 'Pertanyaan Berikutnya ➡️'}
            </button>
          </div>
        </div>
      ) : (
        /* RESULT SCORE CARD */
        <div className="text-center py-6 space-y-6">
          <div className="bg-gradient-to-b from-purple-100 to-amber-50 p-6 rounded-3xl border-4 border-purple-300 shadow-lg max-w-md mx-auto">
            <Trophy className="w-16 h-16 text-yellow-500 mx-auto animate-bounce" />
            <h3 className="text-2xl font-black text-purple-900 mt-2">HASIL KUIS KAMU</h3>

            <div className="my-4">
              <span className="text-5xl font-black text-purple-700">{calculateScore()}</span>
              <span className="text-sm font-bold text-purple-500"> / 100</span>
            </div>

            <p className="text-xs font-bold text-slate-600">
              {calculateScore() === 100
                ? "Sempurna! Kamu memahami firman Tuhan tentang persahabatan dengan sangat baik! 🌟"
                : calculateScore() >= 60
                ? "Bagus sekali! Kamu sudah jadi teman yang hebat! Teruskan ya! 👍"
                : "Tetap semangat! Yuk baca kembali materinya dan coba lagi! 💪"}
            </p>

            <div className="mt-4 inline-flex items-center space-x-2 bg-yellow-300 border-2 border-yellow-500 px-4 py-2 rounded-full font-black text-amber-900 text-xs">
              <Star className="w-4 h-4 fill-amber-700" />
              <span>+{userAnswers.filter(a => a.isCorrect).length * 2} Bintang Ditambahkan!</span>
            </div>

            <div className="mt-6">
              <button
                onClick={resetKuis}
                className="px-6 py-2.5 bg-purple-600 hover:bg-purple-700 text-white font-black rounded-full text-xs shadow flex items-center gap-2 mx-auto"
              >
                <RefreshCw className="w-4 h-4" /> Ulangi Kuis
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
